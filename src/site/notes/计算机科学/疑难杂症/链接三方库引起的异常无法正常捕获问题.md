\#include "throw_test.hpp"

    try {

# 修复方式

## 懒加载

在编译阶段不链接tts库，在运行时加载。

```C++
// catch_test.cpp
\#include <iostream>
\#include "throw_test.hpp"
\#include <dlfcn.h>

void* handle_ = nullptr;

bool LoadLibtts() {
    // 加载共享库
    handle_ = dlopen("libtts.so", RTLD_LAZY);
    if (!handle_) {
        std::cerr << "LoadLibtts failed: " << dlerror() << std::endl;
        return false;
    }
    return true;
}

int main() {
    LoadLibtts();
    /*
    ....
    */
    return 0;
}
```

  

## C++标准库覆盖

在链接libtts之前先链接c++_shared，原来C++程序是默认静态编译c++标准库的，现在改成动态链接方式。

```C++
cmake_minimum_required(VERSION 3.4.1)
project(mission_center_test)

# 设置 C++ 标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 设置链接选项
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,--unresolved-symbols=ignore-in-shared-libs")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Werror=return-type")

# 添加 libxtts 作为导入的共享库
add_library(libtts SHARED IMPORTED)
set_target_properties(libtts PROPERTIES
    IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/libtts.so
)

# # 添加可执行文件 tts_test
add_executable(tts_test catch_test.cpp)

# 链接库到可执行文件
target_link_libraries(tts_test
    PRIVATE
    c++_shared
    libtts
)
```

  

  

# 原理分析

## 异常无法捕获的几种常见情况

除了未捕获、捕获类型与抛出类型不匹配这些情况外，还有几种比较容易忽视的情况。

### 编译禁止异常

可以通过下面代码来禁止异常，禁止后如果代码中有try…catch、throw将会导致编译错误

```C++
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Werror=return-type -fno-exceptions")
```

### **noexcept函数中抛出异常**

申明throw_runtime_local_from_libxtts为noexcept，但是仍抛出异常：

```C++
void throw_runtime_local_from_libxtts() noexcept {
    throw std::runtime_error("throw_runtime_local_from_libxtts");
}
```

编译运行会崩溃。

```C++
root@v1_ea230_dock:/ # /system/bin/tts_test                                                                                                                             
terminating with uncaught exception of type std::runtime_error: throw_runtime_local_from_libxtts
Aborted (core dumped) 
```

  

### 析构函数中抛出异常

在栈展开的过程中，退出某个块后，块内的局部对象将执行析构函数。因为这些析构函数是自动执行的，所以他们不应该抛出异常。一旦在栈展开的过程中析构函数抛出异常，并且析构函数没有捕获到该异常，则程序将终止。

```C++
class A {
public:
    ~A() {
        try {
            throw "throw 1";
        } catch (...) {
            cout << "catch 1" << endl;
        }
        throw "throw 2";
    }
};

int main(int argc, char* argv[]) {
    try {
        A a;
        throw "throw 3";
    } catch (...) {
        cout << "catch 3" << endl;
    }
	return 0;
}
```

运行结果如下：

```C++

catch 1
libc++abi: terminating due to uncaught exception of type char const*
zsh: abort      ./demo
```
